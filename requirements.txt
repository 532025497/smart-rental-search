# 租房选择系统 — 依赖说明
#
# 核心功能(可行域计算 / 小区检索 / 居住体验扫描 / LLM提取)仅用 Python 标准库(urllib)，
# 不安装任何第三方包也能跑 cli.py 和 demo_*.py。
#
# 以下为可选依赖，按需用 pip 安装：

requests>=2.28          # 豆瓣/小红书采集器 (src/collectors/)
urllib3<2               # 兼容 macOS 系统 Python 3.9 的 LibreSSL
beautifulsoup4>=4.11    # 豆瓣帖子HTML解析
flask>=2.2              # Web UI (app.py)
gunicorn>=21,<24        # 生产服务器 (Render 等云平台)

# 安装全部可选依赖:
#   pip install requests beautifulsoup4 flask
